2. الجداول

CREATE TABLE `roles` (
  `id`          INT AUTO_INCREMENT PRIMARY KEY,
  `name`        VARCHAR(100) NOT NULL,
  `guard_name`  VARCHAR(50)  NOT NULL DEFAULT 'web',
  `description` VARCHAR(255) NULL,
  `is_active`   TINYINT(1)   NOT NULL DEFAULT 1,
  `created_at`  TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at`  TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  UNIQUE KEY `roles_name_guard` (`name`, `guard_name`)
);

CREATE TABLE `permissions` (
  `id`          INT AUTO_INCREMENT PRIMARY KEY,
  `name`        VARCHAR(100) NOT NULL,          -- sales.add
  `guard_name`  VARCHAR(50)  NOT NULL DEFAULT 'web',
  `label`       VARCHAR(150) NOT NULL,          -- إضافة فاتورة مبيعات
  `group_name`  VARCHAR(50)  NOT NULL,          -- المبيعات = قسم القائمة الجانبية
  `sort`        INT NOT NULL DEFAULT 0,
  `created_at`  TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at`  TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  UNIQUE KEY `permissions_name_guard` (`name`, `guard_name`)
);

CREATE TABLE `role_has_permissions` (
  `permission_id` INT NOT NULL,
  `role_id`       INT NOT NULL,
  PRIMARY KEY (`permission_id`, `role_id`),
  KEY `rhp_role` (`role_id`),
  CONSTRAINT `rhp_perm_fk` FOREIGN KEY (`permission_id`) REFERENCES `permissions` (`id`) ON DELETE CASCADE,
  CONSTRAINT `rhp_role_fk` FOREIGN KEY (`role_id`)       REFERENCES `roles` (`id`)       ON DELETE CASCADE
);

users.userrole بيفضل هو الربط بين المستخدم ودوره (دور واحد لكل مستخدم).

3. الكلاس Can والـ session

تسجيل الدخول← Can::login() — query واحدة← $_SESSION['auth']← Can::check() من الـ session 
// classes/Can.php
final class Can
{
    public static function login(mysqli $conn, int $userId): void;
    public static function check(string $permission): bool;
    public static function any(string ...$permissions): bool;
    public static function authorize(string ...$permissions): void; // أي واحدة منهم تكفي، وإلا 403 / JSON
    public static function isSuperAdmin(): bool;
    public static function logout(): void;
}

$_SESSION['auth'] = [
    'user_id'  => 7,
    'role_id'  => 3,
    'db'       => 'kody26',   // لو اتغيّرت قاعدة البيانات في نفس الـ session ← logout
    'is_super' => false,
    'perms'    => ['sales.view' => true, 'sales.add' => true],
];
  • المسؤول الرئيسي (users.id = 1 أو usertype = 2): check() ترجع true دايماً.
  • صلاحية مش موجودة = رفض.
  • دور معطّل وقت الدخول = مفيش صلاحيات.
  • تغيير الصلاحيات يسري بعد إعادة الدخول، وده مكتوب كتنبيه في شاشة تعديل الدور.

4. قواعد الحصر

  1. المجموعة = قسم القائمة الجانبية. شاشة الدور بتعرض الصلاحيات بنفس ترتيب القائمة، فالأدمن بيلاقي الحاجة في نفس المكان اللي بيشوفها فيه.
  2. مفيش صلاحيات للقائمة (sid_*). القسم بيظهر لو المستخدم معاه أي صلاحية جوه المجموعة. ده بيشيل 15 عمود كانوا بيتلخبطوا مع صلاحيات الموديولات.
  3. قاعدة CRUD ثابتة: أي كيان بيتعرض ويتضاف ويتعدّل ويتحذف بياخد دايماً الأربع صلاحيات دول، حتى لو صغير زي الوحدات أو المدن، ومفيش .manage بتجمّعهم:
    الصلاحيةCRUDالاسم في شاشة الدوربتحمي إيه
    X.viewReadعرضشاشة القائمة/التفاصيل + ملفات البحث والتصدير بتاعة الكيان
    X.addCreateإضافةشاشة الإضافة + do/doadd_* + الاستيراد
    X.editUpdateتعديلشاشة التعديل + do/doedit_*
    X.deleteDeleteحذفdo/dodel_* + الاسترجاع والحذف النهائي
    add وedit وdelete من غير view ملهمش معنى، فشاشة الدور بتعلّم view تلقائياً لما تعلّم أي واحدة منهم، وdosave_role.php بيطبّق نفس القاعدة في السيرفر.
  4. اللي مش CRUD (تقارير، عمليات زي التسعير والجرد وتقفيل الشيفت، أعلام زي البيع بالسالب) بتاخد صلاحية فعل واحدة باسم واضح.
  5. كل صلاحية جديدة ليها "مصدر في النقل" من usr_pwrs، عشان محدش يكسب أو يخسر صلاحية يوم النقل:
    • add_sales = بتتدّي للأدوار اللي العمود ده عندهم 1.
    • sid_hr = الشاشة النهارده مش محمية غير بإخفاء القسم، فبتتدّي لمين معاه القسم.
    • الكل = الشاشة النهارده مفتوحة لأي حد، فبتتدّي لكل الأدوار.
    • دور 1 = صلاحية جديدة حساسة، بتتدّي للدور رقم 1 بس (والمسؤول الرئيسي بيعدّيها أصلاً).
